-------Write a C program to take two strings as input. Then concatenate the two strings using function-------



----using string----

#include<stdio.h>
void concatenate(char *str1, char *str2)
{
    int i=0, j=0;
    while(str1[i]!='\0')
    {
        i++;
    }
    while(str2[j]!='\0')
    {
        str1[i] = str2[j];//copy
        i++;
        j++;
    }
    str1[i] = '\0';
}
int main(){
char str1[30], str2[30];
printf("enter first string: ");
gets(str1);
printf("enter second string: ");
gets(str2);
concatenate(str1, str2);
printf("concatenated string: %s\n", str1);
}

-------Write a C program to take a string as input. Then convert each character of the string to uppercase using function------


----using string-----


#include<stdio.h>
int stringconvert(char str[])
{
    int i=0;
    if(str[0]>='a' && str[0]<='z')
        str[0] = str[0] - 32;

    for(i=0; str[i]!='\0'; i++)
    {
       if(str[i] == ' ')
       {
        if(str[i+1]>='a' && str[i+1]<='z')
        str[i+1] = str[i+1] - 32;
       }
    }
    return i;
}

int main() {
    char str[100];

    printf("Enter a string: ");
    gets(str);
    int length = stringconvert(str);
    printf("the converted string is: %s\n", str);
    return 0;
}


----using pointers----


#include<stdio.h>
int stringconvert(char *str)
{
    int i=0;
    if((*str)>='a' && (*str)<='z')
        *str = *str - 32;

    for(i=0; *(str+i)!='\0'; i++)
    {
       if(*(str+i) == ' ')
       {
        if(*(str+i)>='a' && *(str+i)<='z')
        *(str+i) = *(str+i) - 32;
       }
    }
    return i;
}

int main() {
    char str[100];

    printf("Enter a string: ");
    gets(str);
    int length = stringconvert(str);//uses return value
    printf("the converted string is: %s\n", str);
    return 0;
}

-------Write a C program to find the frequency of each character present in a string using function-------


-----using string-----


#include<stdio.h>
 int countFrequency(char str[], int freq[256])
{
    int i;
    for(int i=0; str[i]!='\0'; i++)
    {
        unsigned char ch = str[i];
        freq[ch]++;
    }
}
int main() {
    char str[100];
    int freq[256] = {0};
    printf("Enter a string: ");
    gets(str);

    countFrequency(str, freq);

    printf("character frequency:\n");
    for(int i=0; i<256; i++)
    {
        if(freq[i]!='\0')
        {
            printf("'%c' : %d\n", i, freq[i]);
        }
    }
}



------using pointers-----


#include<stdio.h>
 int countFrequency(char *str, int *freq)
{
    int i;
    for(int i=0; *(str+i)!='\0'; i++)
    {
        unsigned char ch = *(str+i);
        (*(freq+ch))++;
    }
}
 int main() {
    char str[100];
    char *p = str;
    int freq[256] = {0};
    int *f = freq;
    printf("Enter a string: ");
    gets(p);

    countFrequency(p, f);

    printf("character frequency:\n");
    for(int i=0; i<256; i++)
    {
        if(*(f+i)!='\0')
        {
            printf("'%c' : %d\n", i, *(f+i));
        }
    }
}




------Write a C program to find out if a string is a substring of another string or not using function------


-----using string-----


#include<stdio.h>
void isSubstring(char str[], char sub[])
{
    int j=0, found;
    for(int i=0; str[i]!='\0'; i++)
    {
         for(j=0; sub[j]!='\0'; j++)
         {
             if(str[i+j]!= sub[j])
                break;//mismatch found
         }
         if(sub[j] == '\0')
         {
             found = 1;
             break;
         }
    }

    if (found == 1) {
        printf("Substring found!\n");
    } else {
        printf("Substring not found.\n");
    }
}
 int main() {
    char str[50], sub[50];
    printf("Enter a string: ");
    gets(str);
    printf("Enter a string: ");
    gets(sub);
    isSubstring(str, sub);
}
